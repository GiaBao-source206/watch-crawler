44def get_json(url, params=None, retries=5):
45    """GET JSON co retry + nghi giua cac lan goi de tranh bi chan (HTTP 429)."""
46    for i in range(retries):
47        try:
48            r = SESSION.get(url, params=params, timeout=20)
49            if r.status_code == 200:
50                return r.json()
51            print(f"  ! HTTP {r.status_code} {url}")
52            if r.status_code == 429:  # bi gioi han toc do -> cho lau hon
53                time.sleep(int(r.headers.get("Retry-After", 0) or 0) or 15 * (i + 1))
54                continue
55        except Exception as e:
56            print(f"  ! loi {e}")
57        time.sleep(1.5 * (i + 1))
58    return None
