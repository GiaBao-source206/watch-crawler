127# --------------------------- WATCHSTORE (Shopify) ---------------------------
128def crawl_watchstore(limit):
129    rows, page = [], 1
130    print("[WatchStore] https://www.watchstore.vn/products.json")
131    while len(rows) < limit:
132        data = get_json("https://www.watchstore.vn/products.json", {"limit": 250, "page": page})
133        products = (data or {}).get("products", [])
134        if not products:
135            break
136        print(f"  trang {page}: {len(products)} san pham")
137        for p in products:
138            if len(rows) >= limit:
139                break
140            v = p["variants"][0]
141            price = float(v.get("price") or 0)
142            if price <= 0 or not p.get("images"):
143                continue  # bo san pham chua co gia / anh
144            compare = float(v.get("compare_at_price") or 0)
145            regular, sale = (compare, price) if compare > price else (price, None)
146            ptype = p.get("product_type") or ""
147            if not ptype:  # Shopify khong co loai -> suy ra tu ten san pham
148                t = p["title"].lower()
149                ptype = "Đồng hồ nữ" if "nữ" in t else "Đồng hồ nam" if "nam" in t else "Khác"
150            attrs = {"Thương hiệu": p.get("vendor", "")}
151            for o in p.get("options", []):
152                if o["name"] != "Title":
153                    attrs[o["name"]] = " | ".join(o["values"])
154            body = p.get("body_html") or ""
155            row = make_row(
156                sku=v.get("sku") or f"WS-{p['id']}", name=p["title"], desc=body,
157                short=clean_text(body)[:200], regular=regular, sale=sale,
158                category=f"Đồng hồ > {ptype}" if ptype != "Khác" else "Đồng hồ",
159                tags=[p.get("vendor"), "WatchStore"],
160                images=[im["src"].split("?")[0] for im in p["images"]],
161                url=f"https://www.watchstore.vn/products/{p['handle']}", attrs=attrs)
162            rows.append(row)
163            print(f"   + [{len(rows):3}] {p['title'][:60]} - {price:,.0f}đ")
164        page += 1
165    return rows
