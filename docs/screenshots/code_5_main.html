168def save_csv(rows, path):
169    seen, uniq = set(), []
170    for r in rows:  # loai trung SKU
171        if r["SKU"] not in seen:
172            seen.add(r["SKU"])
173            uniq.append(r)
174    with open(path, "w", newline="", encoding="utf-8-sig") as f:
175        w = csv.DictWriter(f, fieldnames=COLUMNS, quoting=csv.QUOTE_MINIMAL)
176        w.writeheader()
177        w.writerows(uniq)
178    print(f"\n==> Da ghi {len(uniq)} san pham vao {path}")
