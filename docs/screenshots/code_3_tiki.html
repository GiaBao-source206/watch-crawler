 88def crawl_tiki(category_id, limit):
 89    cat_name = TIKI_CATEGORIES.get(category_id, f"Tiki {category_id}")
 90    rows, page = [], 1
 91    print(f"[Tiki] Danh muc {category_id} - {cat_name}")
 92    while len(rows) < limit:
 93        data = get_json("https://tiki.vn/api/personalish/v1/blocks/listings",
 94                        {"limit": 40, "category": category_id, "page": page})
 95        items = (data or {}).get("data", [])
 96        if not items:
 97            break
 98        print(f"  trang {page}: {len(items)} san pham")
 99        for it in items:
100            if len(rows) >= limit:
101                break
102            d = get_json(f"https://tiki.vn/api/v2/products/{it['id']}")
103            if not d or not d.get("price"):
104                continue
105            brand = (d.get("brand") or {}).get("name", "")
106            attrs = {"Thương hiệu": brand} if brand else {}
107            for grp in d.get("specifications") or []:
108                for a in grp.get("attributes", []):
109                    if a.get("code") in ("origin", "brand_country", "material", "water_resistance"):
110                        attrs[a["name"]] = clean_text(a["value"])
111            imgs = [im.get("base_url") for im in d.get("images", []) if im.get("base_url")]
112            regular = d.get("original_price") or d.get("list_price") or d["price"]
113            row = make_row(
114                sku=f"TIKI-{d['id']}", name=d["name"],
115                desc=d.get("description", ""),
116                short=d.get("short_description", ""),
117                regular=regular, sale=d["price"],
118                category=f"Đồng hồ > {cat_name}", tags=[brand, "Tiki"],
119                images=imgs, url=f"https://tiki.vn/{d.get('url_path', '')}", attrs=attrs)
120            rows.append(row)
121            print(f"   + [{len(rows):3}] {d['name'][:60]} - {d['price']:,}đ")
122            time.sleep(0.3)
123        page += 1
124    return rows
