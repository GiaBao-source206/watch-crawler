66def make_row(sku, name, desc, short, regular, sale, category, tags, images, url, attrs):
67    row = dict.fromkeys(COLUMNS, "")
68    row.update({
69        "Type": "simple", "SKU": sku, "Name": name, "Published": 1, "Is featured?": 0,
70        "Visibility in catalog": "visible", "Short description": short, "Description": desc,
71        "Tax status": "taxable", "In stock?": 1, "Stock": "", "Backorders allowed?": 0,
72        "Sold individually?": 0, "Allow customer reviews?": 1,
73        "Regular price": int(regular) if regular else "",
74        "Sale price": int(sale) if sale and regular and sale < regular else "",
75        "Categories": category, "Tags": ", ".join(t for t in tags if t),
76        # WordPress chi nhan jpg/png/webp -> bo anh .avif, .gif...
77        "Images": ", ".join([u for u in images if re.search(r"\.(jpe?g|png|webp)$", u, re.I)][:5]), "External URL": url, "Position": 0,
78    })
79    for i, (k, v) in enumerate(list(attrs.items())[:3], 1):
80        row[f"Attribute {i} name"] = k
81        row[f"Attribute {i} value(s)"] = str(v).replace(",", " ")
82        row[f"Attribute {i} visible"] = 1
83        row[f"Attribute {i} global"] = 1
84    return row
